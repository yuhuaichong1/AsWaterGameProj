// ------------------------------SDK插入开始------------------------------
task modifyUnityActivity {
    doLast {
        def file = file("src/main/java/com/unity3d/player/UnityPlayerActivity.java")
        if (!file.exists()) {
            println "UnityPlayerActivity.java 文件不存在！请检查路径。"
            return
        }

        def content = file.text

        // 添加导入语句（如果不存在）
        if (!content.contains("import com.abc.sdk.bridge.BridgeImpl;")) {
            content = content.replaceFirst(
                    /(package com\.unity3d\.player;)/,
                    '$1\nimport com.abc.sdk.bridge.BridgeImpl;'
            )
            println "已添加 import com.abc.sdk.bridge.BridgeImpl;"
        }

        if (!content.contains("import com.unity3d.player.io.Bridge;")) {
            content = content.replaceFirst(
                    /(import com\.abc\.sdk\.bridge\.BridgeImpl;)/,
                    '$1\nimport com.unity3d.player.io.Bridge;'
            )
            println "已添加 import com.unity3d.player.io.Bridge;"
        }

        // 添加方法调用（如果不存在）
        // Unity 6 使用 mUnityPlayer.getFrameLayout().requestFocus(); 旧版为 mUnityPlayer.requestFocus();
        if (!content.contains("Bridge.getInstance().addIAndroid2Other(BridgeImpl.getInstance());")) {
            if (content.contains("mUnityPlayer.getFrameLayout().requestFocus();")) {
                content = content.replace(
                        "mUnityPlayer.getFrameLayout().requestFocus();",
                        "mUnityPlayer.getFrameLayout().requestFocus();\n        Bridge.getInstance().addIAndroid2Other(BridgeImpl.getInstance());"
                )
                println "已添加 Bridge.getInstance().addIAndroid2Other(BridgeImpl.getInstance());"
            } else if (content.contains("mUnityPlayer.requestFocus();")) {
                content = content.replace(
                        "mUnityPlayer.requestFocus();",
                        "mUnityPlayer.requestFocus();\n        Bridge.getInstance().addIAndroid2Other(BridgeImpl.getInstance());"
                )
                println "已添加 Bridge.getInstance().addIAndroid2Other(BridgeImpl.getInstance());"
            } else {
                println "未找到 requestFocus 插入点，跳过 Bridge 注入"
            }
        }

        // 写入修改后的内容
        file.text = content
        println "UnityPlayerActivity.java 修改完成！"
    }
}

// 在编译前执行修改任务
preBuild.dependsOn modifyUnityActivity
// ------------------------------SDK插入结束------------------------------
